1. DEPLOYMENT REQUIREMENTS
═══════════════════════════════════════════════════════════════════════════════

For IoT devices to communicate with the system, the backend must be accessible
on the network.

OPTION A: Local Network (Single Building)
──────────────────────────────────────────
- Run backend on a computer connected to same WiFi/LAN
- Get IP address:
  Windows: ipconfig | findstr "IPv4"
  Linux/Mac: ifconfig | grep "inet "
- Example: 192.168.0.101:5000

OPTION B: Cloud Server (Multiple Locations)
──────────────────────────────────────────────
- Deploy backend to cloud (AWS, Azure, Heroku, DigitalOcean)
- Example: https://waste-api.example.com:5000

OPTION C: Docker Container
─────────────────────────────
- Create Dockerfile for backend
- Run with: docker run -p 5000:5000 waste-classifier
- Deploy to Docker Hub, Kubernetes, etc.

2. API ENDPOINT FOR IoT DEVICES
═══════════════════════════════════════════════════════════════════════════════

Main prediction endpoint:

  POST http://<SERVER>:5000/api/predict

REQUIRED:
  - image (multipart file) - JPEG/PNG image of waste
  - source (form data) - Device identifier (e.g., "iot_bin_01", "camera_floor_2")

RESPONSE:
  {
    "success": true,
    "prediction": "plastic",              # Primary classification
    "confidence": 97.5,                   # Percentage
    "top_predictions": [
      {"class": "plastic", "confidence": 0.975},
      {"class": "paper", "confidence": 0.018},
      {"class": "cardboard", "confidence": 0.007}
    ],
    "image_id": 42,
    "prediction_id": 128
  }

C. JavaScript/Node.js (IoT Device with Node)
──────────────────────────────────────────────

const FormData = require('form-data');
const fs = require('fs');
const axios = require('axios');

async function classifyWaste(imagePath, deviceId) {
  const form = new FormData();
  form.append('image', fs.createReadStream(imagePath));
  form.append('source', deviceId);

  try {
    const response = await axios.post(
      'http://192.168.0.101:5000/api/predict',
      form,
      { headers: form.getHeaders() }
    );
    
    console.log('Classification:', response.data.prediction);
    console.log('Confidence:', response.data.confidence + '%');
    
    return response.data;
  } catch (error) {
    console.error('Error:', error.message);
  }
}

// Usage
classifyWaste('waste_image.jpg', 'iot_sensor_warehouse_01');


D. C++ (Embedded Systems)
──────────────────────────

#include <curl/curl.h>
#include <fstream>
#include <iostream>

size_t WriteCallback(void* contents, size_t size, size_t nmemb, std::string* s) {
    s->append((char*)contents, size * nmemb);
    return size * nmemb;
}

int classifyWaste(const std::string& imagePath, const std::string& deviceId) {
    CURL* curl = curl_easy_init();
    if (!curl) return 1;

    std::string readBuffer;
    struct curl_httppost* formpost = nullptr;
    struct curl_httppost* lastptr = nullptr;

    // Add image file
    curl_formadd(&formpost, &lastptr,
                 CURLFORM_COPYNAME, "image",
                 CURLFORM_FILE, imagePath.c_str(), NULL);
    
    // Add device source
    curl_formadd(&formpost, &lastptr,
                 CURLFORM_COPYNAME, "source",
                 CURLFORM_COPYCONTENTS, deviceId.c_str(), NULL);

    curl_easy_setopt(curl, CURLOPT_URL, "http://192.168.0.101:5000/api/predict");
    curl_easy_setopt(curl, CURLOPT_HTTPPOST, formpost);
    curl_easy_setopt(curl, CURLOPT_WRITEFUNCTION, WriteCallback);
    curl_easy_setopt(curl, CURLOPT_WRITEDATA, &readBuffer);

    CURLcode res = curl_easy_perform(curl);
    
    if (res != CURLE_OK) {
        std::cerr << "curl_easy_perform() failed: " << curl_easy_strerror(res) << std::endl;
    } else {
        std::cout << "Response: " << readBuffer << std::endl;
    }

    curl_slist_free_all(NULL);
    curl_easy_cleanup(curl);
    
    return (res == CURLE_OK) ? 0 : 1;
}

int main() {
    return classifyWaste("waste.jpg", "iot_robot_01");
}

6. BATCH PROCESSING (For High Volume)
═══════════════════════════════════════════════════════════════════════════════

Instead of sending one image at a time, batch process for efficiency:

Python example (Process 1000 images):

  import requests
  import os
  from concurrent.futures import ThreadPoolExecutor, as_completed
  
  API_URL = "http://192.168.0.101:5000/api/predict"
  IMAGE_DIR = "/path/to/waste/images"
  DEVICE_ID = "batch_processor_01"
  
  def send_image(image_path):
      with open(image_path, 'rb') as img:
          files = {'image': img}
          data = {'source': DEVICE_ID}
          response = requests.post(API_URL, files=files, data=data)
      return response.json()
  
  # Parallel processing (4 concurrent requests)
  with ThreadPoolExecutor(max_workers=4) as executor:
      images = [os.path.join(IMAGE_DIR, f) for f in os.listdir(IMAGE_DIR)]
      futures = [executor.submit(send_image, img) for img in images]
      
      for future in as_completed(futures):
          result = future.result()
          print(f"Prediction: {result['prediction']}")
  
  # Get summary
  summary_response = requests.get(
    "http://192.168.0.101:5000/api/analytics/breakdown"
  )
  print(summary_response.json())

7. ERROR HANDLING & RESILIENCE
═══════════════════════════════════════════════════════════════════════════════

Network failures are common in IoT. Implement retry logic:

Python retry example:

  import time
  import requests
  from requests.adapters import HTTPAdapter
  from urllib3.util.retry import Retry
  
  def create_session():
      session = requests.Session()
      
      # Retry 3 times on failure
      retry_strategy = Retry(
          total=3,
          status_forcelist=[429, 500, 502, 503, 504],
          method_whitelist=["POST", "GET"],
          backoff_factor=1  # Wait 1s, 2s, 4s
      )
      adapter = HTTPAdapter(max_retries=retry_strategy)
      session.mount("http://", adapter)
      session.mount("https://", adapter)
      
      return session
  
  session = create_session()
  response = session.post(
      API_URL,
      files=files,
      data=data,
      timeout=10
  )

Offline fallback example:

  import json
  import sqlite3
  
  # When network fails, store locally
  try:
      response = requests.post(API_URL, ..., timeout=5)
  except requests.exceptions.RequestException:
      print("Network error - caching for later")
      
      # Save to local SQLite
      conn = sqlite3.connect('offline_queue.db')
      cursor = conn.cursor()
      cursor.execute(
          'INSERT INTO pending (image_path, timestamp) VALUES (?, ?)',
          ('waste.jpg', time.time())
      )
      conn.commit()
      conn.close()
  
  # When connection restored, process queue
  # Periodic background job syncs offline_queue.db with server

10. TROUBLESHOOTING
═══════════════════════════════════════════════════════════════════════════════

❌ Connection refused (http://192.168.0.101:5000)
   ✅ Check: Backend is running (python app.py)
   ✅ Check: IP is correct (ipconfig)
   ✅ Check: Port 5000 is not blocked by firewall
   ✅ Try: Restart Flask server

❌ 404 /api/predict not found
   ✅ Check: Using correct endpoint (not /predict, must be /api/predict)
   ✅ Check: Using POST method, not GET

❌ 413 Request entity too large
   ✅ Check: Image file size < 16MB
   ✅ Solution: Compress/resize image before sending

❌ Timeout (no response after 30 seconds)
   ✅ Check: Server CPU/memory not maxed out
   ✅ Check: Network bandwidth sufficient
   ✅ Solution: Use batch processing with queuing

❌ Model not loaded / 97% cardboard predictions
   ✅ Check: best_model.h5 exists in backend/
   ✅ Check: class_names.json exists and has 17 entries
   ✅ Check: App.py logs for model load errors
